cmake_minimum_required(VERSION 3.14)
project(agent-libuv C)

set(CMAKE_C_STANDARD 99)
include(FetchContent)

# libuv and json-c: use system copies if present, otherwise build from source.
find_package(libuv CONFIG QUIET)
if(NOT TARGET uv AND NOT TARGET libuv::uv)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(LIBUV QUIET IMPORTED_TARGET libuv)
  endif()
  if(TARGET PkgConfig::LIBUV)
    add_library(agent_uv INTERFACE)
    target_link_libraries(agent_uv INTERFACE PkgConfig::LIBUV)
  else()
    message(STATUS "libuv not found, fetching")
    set(LIBUV_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    set(LIBUV_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(libuv GIT_REPOSITORY https://github.com/libuv/libuv.git GIT_TAG v1.48.0)
    FetchContent_MakeAvailable(libuv)
    add_library(agent_uv INTERFACE)
    target_link_libraries(agent_uv INTERFACE uv_a)
  endif()
else()
  add_library(agent_uv INTERFACE)
  target_link_libraries(agent_uv INTERFACE $<IF:$<TARGET_EXISTS:libuv::uv>,libuv::uv,uv>)
endif()

find_package(json-c CONFIG QUIET)
if(TARGET json-c::json-c)
  add_library(agent_json INTERFACE)
  target_link_libraries(agent_json INTERFACE json-c::json-c)
else()
  message(STATUS "json-c not found, fetching")
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(BUILD_APPS OFF CACHE BOOL "" FORCE)
  set(DISABLE_WERROR ON CACHE BOOL "" FORCE)
  FetchContent_Declare(json-c GIT_REPOSITORY https://github.com/json-c/json-c.git GIT_TAG json-c-0.17-20230812)
  FetchContent_MakeAvailable(json-c)
  add_library(agent_json INTERFACE)
  target_link_libraries(agent_json INTERFACE json-c)
  # The source tree has flat headers; stage them so <json-c/json.h> resolves.
  set(JSONC_STAGE ${CMAKE_BINARY_DIR}/jsonc_include/json-c)
  file(GLOB JSONC_HEADERS ${json-c_SOURCE_DIR}/*.h ${json-c_BINARY_DIR}/*.h)
  file(COPY ${JSONC_HEADERS} DESTINATION ${JSONC_STAGE})
  target_include_directories(agent_json INTERFACE ${CMAKE_BINARY_DIR}/jsonc_include)
endif()

# QuickJS-NG is small and has no system packaging worth depending on: always vendor it.
set(QJS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.28)
  set(_exclude EXCLUDE_FROM_ALL)
endif()
FetchContent_Declare(quickjs GIT_REPOSITORY https://github.com/quickjs-ng/quickjs.git
                     GIT_TAG v0.17.0 ${_exclude})
FetchContent_MakeAvailable(quickjs)

# libcurl provides HTTPS; it is available on every mainstream platform.
find_package(CURL REQUIRED)

# Embed the JS prelude into a header.
set(PRELUDE_JS ${CMAKE_SOURCE_DIR}/src/prelude.js)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PRELUDE_JS})
file(READ ${PRELUDE_JS} PRELUDE_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," PRELUDE_BYTES "${PRELUDE_HEX}")
file(WRITE ${CMAKE_BINARY_DIR}/gen/prelude_gen.h
     "static const unsigned char prelude_js[] = {${PRELUDE_BYTES}0x00};\n")

add_executable(agent src/agent.c src/http.c src/proc.c src/script.c)
target_include_directories(agent PRIVATE ${CMAKE_BINARY_DIR}/gen)
target_link_libraries(agent PRIVATE agent_uv agent_json qjs CURL::libcurl)
if(NOT MSVC)
  target_compile_options(agent PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
install(TARGETS agent RUNTIME DESTINATION bin)
